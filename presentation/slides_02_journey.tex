\begin{frame}{Sub-1: Baseline --- What It Did and Why It Was Slow}
\begin{columns}[T]
\column{0.52\textwidth}
\small\textbf{Approach:}
\begin{itemize}\small
  \item Per-expert loop: 32 Python iterations
  \item Dequant FP8$\to$FP32 (full weight matrix) before each GEMM
  \item Separate SwiGLU kernel after GEMM1
  \item 96 kernel launches for dispatch (3 per expert)
\end{itemize}
\medskip
\textbf{B200 Results (19/19):}
\scriptsize
\begin{tabular}{rrl}
\toprule
Seq & Time & Speedup \\
\midrule
1     & 2.57 ms  & 4.35$\times$ \\
7     & 4.14 ms  & 2.86$\times$ \\

901   & 20.56 ms & 1.04$\times$ \\
14107 & 44.07 ms & 1.03$\times$ \\
\bottomrule
\end{tabular}

\column{0.45\textwidth}
\begin{alertblock}{\small Why Large-T Was Near 1$\times$}
\scriptsize
\textbf{Weight materialize cost:}\\
$32\;\text{experts} \times 112\;\text{MB (W1 FP32)}$\\
$= 3.6\;\text{GB}$ written to HBM per forward\\
At $8\;\text{TB/s} = 0.45\;\text{ms}$ minimum\\
\textit{(just for weight copies!)}

\medskip
\textbf{Dispatch cost:}\\
96 launches $\times$ 5--15\,$\mu$s $= \sim\!0.5\;\text{ms}$\\
Dominates for small-T workloads.
\end{alertblock}
\end{columns}
\end{frame}

\begin{frame}[fragile]{Dispatch: 96 Kernel Launches $\to$ 4}
\begin{columns}[T]
\column{0.48\textwidth}
\begin{alertblock}{Sub-1 Approach (96 launches)}
\scriptsize
\begin{lstlisting}
for expert in range(32):
  mask = (local_idx == expert) & valid  # launch
  if mask.any():                         # launch
    t_ids = mask.nonzero()               # launch
    # 3 launches x 32 experts = 96
\end{lstlisting}
Each \texttt{nonzero}/\texttt{any} = 5--15\,$\mu$s.\\
Total: $\approx 300\,\mu$s wasted \textbf{per call}.
\end{alertblock}

\column{0.48\textwidth}
\begin{exampleblock}{Sub-3 Approach (4 launches)}
\scriptsize
\begin{lstlisting}
# 1. all valid pairs in one shot
all_valid = nonzero(valid_local)     # 1 launch
# 2. sort by expert id
order = argsort(expert_ids)          # 1 launch
# 3. find expert boundaries
unique_e, counts = \
  unique_consecutive(sorted_e)       # 1 launch
# loop with Python index math = free
\end{lstlisting}
Result: 10--34\% speedup across ALL workloads.\\
Small-T gained most (dispatch = 60\% of time).
\end{exampleblock}
\end{columns}

\begin{frame}{Sub-3: GEMM1 + SwiGLU Fusion}
\begin{columns}[T]
\column{0.52\textwidth}
\small\textbf{Two changes combined:}
\begin{enumerate}\small
  \item Merged gate+up GEMM and SwiGLU into \textbf{one Triton kernel}
  \item Replaced 96-launch dispatch with 4-launch table
\end{enumerate}

\medskip\textbf{B200 Results (19/19):}
\scriptsize\begin{tabular}{rrl}
\toprule
Seq & Time & Speedup \\ \midrule
1     & 1.716 ms  & 6.50$\times$ \\
7     & 3.172 ms  & 3.73$\times$ \\
15    & 2.626 ms  & 4.38$\times$ \\
32    & 9.082 ms  & 1.58$\times$ \\
14107 & 24.363 ms & 1.85$\times$ \\
\bottomrule
\end{tabular}

\column{0.45\textwidth}
\begin{block}{\small GEMM1+SwiGLU Fusion Benefit}
\scriptsize
\textbf{Before:}\\
GEMM1 writes $[T_k, 4096]$ FP32 to HBM\\
SwiGLU reads it back\\
$\Rightarrow T_k \times 4096 \times 8$ bytes round-trip

\medskip
\textbf{After (fused):}\\
\texttt{gate\_acc}, \texttt{up\_acc} stay in registers\\
SwiGLU in epilogue, no store before it\\
$\Rightarrow$ saves \textbf{8.4 MB/expert} at $T_k=256$\\
$\Rightarrow$ removes 1 kernel launch per expert
\end{block}

\begin{exampleblock}{\small A-tile Sharing}
\scriptsize
A tile loaded ONCE per K-block.\\
Used for both gate dot and up dot.\\
Halves A-read bandwidth in GEMM1.
\end{exampleblock}
\end{columns}
\end{frame}

\begin{frame}[fragile]{Sub-6: Route-Weight Fused into GEMM2 Epilogue}
\begin{columns}[T]
\column{0.52\textwidth}
\small\textbf{The idea:}\\
\smallskip
Each expert output is multiplied by route weight $w_e$ before scatter-add.
Previously a separate kernel after GEMM2.

\medskip
\textbf{Fix:} multiply at store time in GEMM2 --- free in registers:
\smallskip
\scriptsize
\begin{lstlisting}
# Triton GEMM2 epilogue:
rw = tl.load(route_w_ptr + offs_m)
tl.store(o_ptr + ..., acc * rw[:, None])
\end{lstlisting}

One scalar load per output row. Zero extra HBM traffic.

\column{0.45\textwidth}
\begin{block}{\small B200 Impact}
\scriptsize
\begin{tabular}{rrl}
\toprule
Seq & Before & After \\ \midrule
901   & 10.06 ms & \textbf{8.70 ms} ($+16\%$) \\
11948 & 17.65 ms & \textbf{14.60 ms} ($+17\%$) \\
14107 & 24.36 ms & \textbf{19.28 ms} ($+21\%$) \\
\bottomrule
\end{tabular}
\end{block}

\begin{exampleblock}{\small Why Large-T Gains Most}
\scriptsize
GEMM2 output: $T_k \times 7168 \times 4\;\text{B}$.\\
At $T_k=256$: 7.3 MB saved per expert.\\
$\times 32$ experts $= 234\;\text{MB}$ fewer r/w.
\end{exampleblock}
\end{columns}
\end{frame}

\begin{frame}{Sub-9: Lossless FP8$\to$BF16 Promotion + BF16 Tensor Cores}
\begin{columns}[T]
\column{0.5\textwidth}
\small\textbf{Key insight:}\\
FP8 E4M3 has \textbf{4 significant bits}.\\
BF16 has \textbf{7 significant bits}.\\
$\Rightarrow$ Promoting FP8$\to$BF16 is \textbf{lossless}.

\medskip
\textbf{Old approach (Sub-3 through Sub-6):}
\scriptsize
\begin{enumerate}
  \item $a_\text{fp8} \times S_a \to$ FP32 (dequant)
  \item FP32 $\to$ BF16 \textcolor{deepred}{\textbf{[LOSSY: 16 bits of scale lost]}}
  \item BF16 dot $\to$ FP32 accumulator
\end{enumerate}

\medskip\small
\textbf{Sub-9 approach:}
\scriptsize
\begin{enumerate}
  \item $a_\text{fp8} \to$ BF16 directly \textcolor{deepgreen}{\textbf{[LOSSLESS]}}
  \item BF16 $\times$ BF16 dot $\to$ FP32 raw sum
  \item Apply $S_a \times S_w$ in FP32 \textbf{post-dot}
\end{enumerate}

\column{0.47\textwidth}
\begin{exampleblock}{\small Mathematical Equivalence}
\scriptsize
Before: $\sum_k \bigl(a_\text{fp8}[k]\!\cdot\!S_a\bigr)_\text{bf16} \cdot w_\text{fp8}[k]$\\[4pt]
After:\; $S_a \cdot S_w \cdot \sum_k a_\text{fp8}[k] \cdot w_\text{fp8}[k]$

\medskip
Scale factored \emph{out} of the inner dot.\\
Inner sum uses raw FP8 bits in BF16 containers.\\
Scale applied once in full FP32.
\end{exampleblock}

\begin{block}{\small Also: FP8 Bulk Gather}
\scriptsize
Gather tokens \textbf{before} dequant:\\
FP8: 1 byte/elem vs FP32: 4 bytes/elem\\
$\Rightarrow 4\times$ less gather bandwidth\\
Saves 5.4 MB/expert at $T_k=256$.
\end{block}
\end{columns}
\end{frame}
