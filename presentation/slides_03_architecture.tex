\begin{frame}{Sub-9 Architecture: Key Decisions at a Glance}
\begin{columns}[T]
\column{0.5\textwidth}
\begin{exampleblock}{What runs in each stage}
\small
\begin{tabular}{ll}
\toprule
\textbf{Stage} & \textbf{Where} \\
\midrule
Routing (sigmoid, topk) & PyTorch (vectorized) \\
Dispatch (sort, gather) & PyTorch (4 ops) \\
GEMM1 + SwiGLU & \textbf{Triton} (BF16 TC) \\
GEMM2 + route-wt & \textbf{Triton} (TF32 TC) \\
Scatter-add & PyTorch \texttt{index\_add\_} \\
\bottomrule
\end{tabular}
\end{exampleblock}

\begin{block}{\small Precision Chain}
\scriptsize
FP8 tokens $\xrightarrow{\text{lossless}}$ BF16 $\xrightarrow{\text{dot}}$ FP32 raw\\
$\xrightarrow{\times S_a S_w}$ FP32 accumulator\\
$\xrightarrow{\text{accum}}$ FP32 output\\
$\xrightarrow{\text{cast}}$ BF16 final output\\[3pt]
\textbf{Rule:} BF16 safe only for FP8-origin data.\\
Any FP32-computed value must stay FP32.
\end{block}

\column{0.47\textwidth}
\begin{block}{\small Memory Traffic Per Expert ($T_k=256$)}
\scriptsize
\begin{tabular}{lr}
\toprule
\textbf{Operation} & \textbf{Traffic} \\
\midrule
FP8 gather (1 byte/elem) & 1.8 MB \\
W1 FP8 read (28 MB) & 28 MB \\
W2 FP8 read (14 MB) & 14 MB \\
GEMM1 A-tiles (FP8) & 7.2 MB \\
GEMM2 C-tiles (FP32) & 29 MB \\
Output write (FP32) & 7.3 MB \\
\midrule
\textbf{Total} & $\sim$87 MB \\
\bottomrule
\end{tabular}
\smallskip

Dominant cost: weight reads + C-tile loads.\\
FP8 gather saved 5.4 MB vs FP32.\\
Epilogue fusion saved 7.3 MB output re-read.
\end{block}
\end{columns}
\end{frame}
